cmake_minimum_required(VERSION 3.25)
project(K09BinSortReuse LANGUAGES CXX)
find_package(Kokkos CONFIG REQUIRED)
add_executable(k09_binsort_reuse binsort_reuse.cpp)
target_compile_features(k09_binsort_reuse PRIVATE cxx_std_20)
target_link_libraries(k09_binsort_reuse PRIVATE Kokkos::kokkos)
target_compile_options(k09_binsort_reuse PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address)
target_link_options(k09_binsort_reuse PRIVATE -fsanitize=address)

add_executable(k09_binsort_offset binsort_offset.cpp)
target_compile_features(k09_binsort_offset PRIVATE cxx_std_20)
target_link_libraries(k09_binsort_offset PRIVATE Kokkos::kokkos)
target_compile_options(k09_binsort_offset PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address)
target_link_options(k09_binsort_offset PRIVATE -fsanitize=address)
